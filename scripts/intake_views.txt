const CoachIntakeView = ({ coachId, onNavigateHome }) => {
  const [profile, setProfile] = g.useState(null);
  const [loading, setLoading] = g.useState(true);
  const [selectedTier, setSelectedTier] = g.useState("pro_coaching");
  const [athleteName, setAthleteName] = g.useState("");
  const [athleteEmail, setAthleteEmail] = g.useState("");
  const [isSubmitting, setIsSubmitting] = g.useState(false);
  const [errorMsg, setErrorMsg] = g.useState(null);

  g.useEffect(() => {
    if (typeof document !== "undefined") {
      document.documentElement.classList.add("athlete-intake-route");
      document.body.classList.add("athlete-intake-route");
      const rootEl = document.getElementById("root");
      if (rootEl) rootEl.classList.add("athlete-intake-route");
      return () => {
        document.documentElement.classList.remove("athlete-intake-route");
        document.body.classList.remove("athlete-intake-route");
        if (rootEl) rootEl.classList.remove("athlete-intake-route");
      };
    }
  }, []);

  g.useEffect(() => {
    let mounted = true;
    fetch("/api/coach/profile?coach_id=" + encodeURIComponent(coachId || "default_coach"))
      .then(res => res.json())
      .then(data => {
        if (mounted && data.profile) {
          setProfile(data.profile);
        }
      })
      .catch(err => console.warn("Failed to load coach profile:", err))
      .finally(() => { if (mounted) setLoading(false); });
    return () => { mounted = false; };
  }, [coachId]);

  const pricingTiers = [
    {
      id: "standard_coaching",
      name: "Standard Performance",
      price_aud: 149,
      description: "Custom weekly training program, workout tracking, and progressive overload calculations."
    },
    {
      id: "pro_coaching",
      name: "Pro Athlete Protocol",
      price_aud: 249,
      description: "Comprehensive customized programming with video form feedback and direct coach messaging."
    },
    {
      id: "elite_1on1",
      name: "Elite 1-on-1 Performance",
      price_aud: 399,
      description: "Fully individualized coaching protocol with 1-on-1 consultations and priority check-ins."
    }
  ];

  const activeTierObj = pricingTiers.find(t => t.id === selectedTier) || pricingTiers[1];

  const handleSubmit = async (e) => {
    if (e && e.preventDefault) e.preventDefault();
    if (!athleteName.trim()) {
      setErrorMsg("Please enter your full name.");
      return;
    }
    if (!athleteEmail.trim() || !athleteEmail.includes("@")) {
      setErrorMsg("Please enter a valid email address.");
      return;
    }

    setErrorMsg(null);
    setIsSubmitting(true);

    try {
      const payload = {
        coach_id: coachId || "default_coach",
        athlete_name: athleteName.trim(),
        athlete_email: athleteEmail.trim().toLowerCase(),
        primary_goal: "Comprehensive Athletic Coaching",
        medical_clearance: "Cleared",
        plan_tier: selectedTier,
        mode: "subscription",
        client_origin: window.location.origin
      };

      const res = await fetch("/api/coach/web-checkout", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload)
      });

      const data = await res.json();
      if (!res.ok || !data.url) {
        throw new Error(data.error || "Failed to create secure checkout session.");
      }

      window.location.href = data.url;
    } catch (err) {
      console.error("Checkout initiation error:", err);
      setErrorMsg(err.message || "An error occurred initiating checkout. Please try again.");
      setIsSubmitting(false);
    }
  };

  const coachName = (profile && profile.display_name) || "Marcus Vance";
  const coachTitle = (profile && profile.title) || "Head of Athletic Performance";
  const coachAvatar = (profile && profile.avatar) || "https://images.unsplash.com/photo-1534528741775-53994a69daeb?w=300&auto=format&fit=crop&q=80";

  return r.jsxs("div", {
    className: "athlete-intake-wrapper w-full min-h-screen h-auto overflow-y-auto overflow-x-hidden bg-[#0A0C10] text-[#F5F5F7] font-sans pb-24 selection:bg-red-600/30",
    style: {
      WebkitOverflowScrolling: "touch",
      minHeight: "100vh",
      height: "auto",
      overflowY: "auto"
    },
    children: [
      // Top Navigation Bar
      r.jsx("header", {
        className: "sticky top-0 z-40 bg-[#0A0C10]/90 backdrop-blur-md border-b border-white/[0.08] px-4 py-3",
        children: r.jsxs("div", {
          className: "max-w-xl mx-auto flex items-center justify-between",
          children: [
            r.jsxs("div", {
              className: "flex items-center gap-2",
              children: [
                r.jsx("span", { className: "w-2 h-2 rounded-full bg-red-600" }),
                r.jsx("span", { className: "font-mono font-bold tracking-widest text-xs uppercase text-white", children: "O1FC ATHLETE INTAKE" })
              ]
            }),
            r.jsx("button", {
              type: "button",
              onClick: onNavigateHome,
              className: "text-[11px] font-mono text-zinc-400 hover:text-white transition-colors cursor-pointer px-2 py-1 rounded-lg hover:bg-white/[0.05]",
              children: "App Home →"
            })
          ]
        })
      }),

      r.jsxs("main", {
        className: "max-w-xl mx-auto px-4 pt-6 space-y-6 pb-12",
        children: [
          // 1. Coach Profile Card (Avatar, Name, Title only)
          r.jsx("div", {
            className: "p-4 sm:p-5 rounded-2xl bg-white/[0.03] border border-white/[0.08] shadow-lg",
            children: r.jsxs("div", {
              className: "flex items-center gap-4",
              children: [
                r.jsx("img", {
                  src: coachAvatar,
                  alt: coachName,
                  className: "w-16 h-16 rounded-2xl object-cover border border-white/10 shadow-md shrink-0"
                }),
                r.jsxs("div", {
                  className: "min-w-0 flex-1",
                  children: [
                    r.jsx("h1", { className: "text-lg sm:text-xl font-bold tracking-tight text-white", children: coachName }),
                    r.jsx("p", { className: "text-xs text-zinc-400 font-mono mt-0.5", children: coachTitle })
                  ]
                })
              ]
            })
          }),

          // 2. Clean Tier Selection
          r.jsxs("div", {
            className: "space-y-3",
            children: [
              r.jsx("h2", { className: "text-xs font-bold uppercase font-mono tracking-wider text-zinc-400", children: "Select Coaching Plan" }),
              r.jsx("div", {
                className: "grid grid-cols-1 sm:grid-cols-3 gap-2.5",
                children: pricingTiers.map(tier => {
                  const isSelected = selectedTier === tier.id;
                  return r.jsxs("button", {
                    key: tier.id,
                    type: "button",
                    onClick: () => setSelectedTier(tier.id),
                    className: "p-4 rounded-xl border text-left transition-all cursor-pointer flex flex-col justify-between " + (isSelected ? "bg-red-600/10 border-red-500 shadow-md shadow-red-500/10 ring-1 ring-red-500" : "bg-white/[0.02] border-white/10 hover:border-white/20"),
                    children: [
                      r.jsxs("div", {
                        children: [
                          r.jsx("div", { className: "text-xs font-bold text-white", children: tier.name }),
                          r.jsxs("div", {
                            className: "mt-1 flex items-baseline gap-1",
                            children: [
                              r.jsx("span", { className: "text-lg font-black text-white font-mono", children: "$" + tier.price_aud }),
                              r.jsx("span", { className: "text-[10px] font-mono text-zinc-400", children: "AUD/mo" })
                            ]
                          })
                        ]
                      }),
                      r.jsx("p", { className: "mt-2.5 text-xs text-zinc-400 leading-relaxed", children: tier.description })
                    ]
                  });
                })
              })
            ]
          }),

          // 3. Minimal Intake Form & Ultra-Clean Checkout Bar
          r.jsxs("form", {
            onSubmit: handleSubmit,
            className: "p-4 sm:p-5 rounded-2xl bg-white/[0.03] border border-white/[0.08] space-y-4 font-mono",
            children: [
              r.jsxs("div", {
                children: [
                  r.jsx("h2", { className: "text-xs font-bold uppercase tracking-wider text-zinc-400", children: "Athlete Details" }),
                  r.jsx("p", { className: "text-[11px] text-zinc-400 mt-0.5", children: "Enter your contact details to begin onboarding with your coach." })
                ]
              }),

              // Full Name & Email
              r.jsxs("div", {
                className: "grid grid-cols-1 sm:grid-cols-2 gap-3",
                children: [
                  r.jsxs("div", {
                    children: [
                      r.jsx("label", { className: "text-[10px] text-zinc-400 uppercase font-bold block mb-1", children: "Full Name *" }),
                      r.jsx("input", {
                        type: "text",
                        required: true,
                        placeholder: "Alex Hunter",
                        value: athleteName,
                        onChange: e => setAthleteName(e.target.value),
                        className: "w-full px-3 py-2.5 rounded-xl bg-black/40 border border-white/10 text-white placeholder-zinc-600 text-xs focus:outline-none focus:border-red-500 focus:ring-1 focus:ring-red-500 transition-all font-sans"
                      })
                    ]
                  }),
                  r.jsxs("div", {
                    children: [
                      r.jsx("label", { className: "text-[10px] text-zinc-400 uppercase font-bold block mb-1", children: "Email Address *" }),
                      r.jsx("input", {
                        type: "email",
                        required: true,
                        placeholder: "alex@athlete.com",
                        value: athleteEmail,
                        onChange: e => setAthleteEmail(e.target.value),
                        className: "w-full px-3 py-2.5 rounded-xl bg-black/40 border border-white/10 text-white placeholder-zinc-600 text-xs focus:outline-none focus:border-red-500 focus:ring-1 focus:ring-red-500 transition-all font-sans"
                      })
                    ]
                  })
                ]
              }),

              // Error display if any
              errorMsg && r.jsx("div", {
                className: "p-3 rounded-xl bg-red-500/10 border border-red-500/30 text-red-400 text-xs font-mono",
                children: errorMsg
              }),

              // 4. Ultra-Clean Checkout Bar
              r.jsxs("div", {
                className: "pt-2 space-y-2.5",
                children: [
                  r.jsxs("div", {
                    className: "flex items-center justify-between text-xs font-mono text-zinc-300 px-0.5",
                    children: [
                      r.jsx("span", { children: "Monthly Investment" }),
                      r.jsx("span", { className: "text-white font-bold text-sm", children: "$" + activeTierObj.price_aud + " AUD" })
                    ]
                  }),

                  r.jsx("button", {
                    type: "submit",
                    disabled: isSubmitting,
                    className: "w-full py-3.5 px-4 rounded-xl bg-red-600 hover:bg-red-500 active:scale-[0.99] disabled:opacity-50 text-white font-bold text-xs uppercase tracking-wider font-mono shadow-lg flex items-center justify-center gap-2 transition-all cursor-pointer",
                    children: isSubmitting
                      ? r.jsxs(r.Fragment, {
                          children: [
                            r.jsx("span", { className: "w-4 h-4 border-2 border-white border-t-transparent rounded-full animate-spin" }),
                            r.jsx("span", { children: "CONNECTING TO CHECKOUT..." })
                          ]
                        })
                      : r.jsx("span", { children: "Proceed to Secure Enrollment" })
                  })
                ]
              })
            ]
          })
        ]
      })
    ]
  });
};

const AthleteWelcomeView = ({ onNavigateHome }) => {
  const [sessionInfo, setSessionInfo] = g.useState(null);
  const [loading, setLoading] = g.useState(true);

  g.useEffect(() => {
    const params = new URLSearchParams(window.location.search);
    const sessionId = params.get("session_id") || "";
    const coachId = params.get("coach_id") || "";
    const clientEmail = params.get("client_email") || "";
    const clientName = params.get("client_name") || "";
    const planTier = params.get("plan_tier") || "";

    const url = "/api/coach/checkout-session?session_id=" + encodeURIComponent(sessionId) + "&coach_id=" + encodeURIComponent(coachId) + "&client_email=" + encodeURIComponent(clientEmail) + "&client_name=" + encodeURIComponent(clientName) + "&plan_tier=" + encodeURIComponent(planTier);

    fetch(url)
      .then(res => res.json())
      .then(data => {
        if (data.success) {
          setSessionInfo(data);
        }
      })
      .catch(err => console.warn("Session check notice:", err))
      .finally(() => setLoading(false));
  }, []);

  const coach = (sessionInfo && sessionInfo.coach) || { display_name: "Coach Marcus Vance", coach_id: "default_coach" };

        className: "sticky top-0 z-40 bg-[#0A0C10]/90 backdrop-blur-md border-b border-white/[0.08] px-4 py-3",
        children: r.jsxs("div", {
          className: "max-w-xl mx-auto flex items-center justify-between",
          children: [
            r.jsxs("div", {
              className: "flex items-center gap-2",
              children: [
                r.jsx("span", { className: "w-2.5 h-2.5 rounded-full bg-emerald-500" }),
                r.jsx("span", { className: "font-mono font-bold tracking-widest text-xs uppercase text-white", children: "O1FC ENROLLMENT CONFIRMED" })
              ]
            }),
            r.jsx("button", {
              type: "button",
              onClick: onNavigateHome,
              className: "text-[11px] font-mono text-zinc-400 hover:text-white transition-colors cursor-pointer px-2 py-1 rounded-lg hover:bg-white/[0.05]",
